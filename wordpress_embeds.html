County Map — Liquid Spill Count

Paste into WordPress Custom HTML block — requires D3 + TopoJSON
<div id="cwp-nm-county-liq-wrap" style="max-width:700px;margin:24px auto;">
<svg id="cwp-nm-county-liq" viewBox="0 0 700 480" width="100%"></svg>
<div style="font-size:11px;color:#9a9590;font-family:'Open Sans',sans-serif;margin-top:4px;">Liquid spills by county (count). Source: NM OCD, 2025.</div>
</div>
<script>
(function(){
  var data={Eddy:598,Lea:547,'San Juan':85,'Rio Arriba':32,Chaves:8,Sandoval:3,Roosevelt:2,Curry:1,Lincoln:1};
  function drawCwpNmLiqMap(){
    Promise.all([
      fetch('https://cdn.jsdelivr.net/npm/topojson-client@3/dist/topojson-client.min.js').then(function(){return window.topojson;}),
      fetch('https://cdn.jsdelivr.net/npm/us-atlas@3/counties-10m.json').then(function(r){return r.json();})
    ]).then(function(results){
      var topo=results[0],us=results[1];
      var w=700,h=480,svg=d3.select('#cwp-nm-county-liq').attr('viewBox','0 0 '+w+' '+h).attr('width','100%');
      var counties=topojson.feature(us,us.objects.counties);
      var nm=counties.features.filter(function(d){return d.id.toString().startsWith('35');});
      var maxVal=Math.max.apply(null,Object.values(data));
      var color=d3.scaleSequential([0,maxVal],d3.interpolate('#e8f4f8','#2f5b74'));
      var proj=d3.geoAlbers().fitExtent([[20,20],[w-20,h-20]],{type:'FeatureCollection',features:nm});
      var path=d3.geoPath(proj);
      var tip=d3.select('body').append('div').style('position','fixed').style('pointer-events','none')
        .style('background','rgba(30,38,36,.92)').style('color','#fff').style('padding','6px 10px')
        .style('border-radius','4px').style('font-family',"'Open Sans',sans-serif").style('font-size','12px')
        .style('white-space','nowrap').style('z-index','99999').style('opacity',0).style('transition','opacity .1s');
      svg.selectAll('path').data(nm).enter().append('path')
        .attr('d',path).attr('fill',function(d){
          var name=d.properties.name,val=data[name]||0;
          return val>0?color(val):'#f5f5f3';
